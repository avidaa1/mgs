function snakeAI(msg){
  const q = msg.toLowerCase().trim();
  // Direct key match
  for(const [key, val] of Object.entries(SNAKE_KB)){
    if(q.includes(key)) return val;
  }
  // Partial match
  const words = q.split(/\s+/);
  for(const [key, val] of Object.entries(SNAKE_KB)){
    const keyWords = key.split(/\s+/);
    if(keyWords.every(kw => q.includes(kw))) return val;
  }
  // Fallback hints
  if(q.includes('who is') || q.includes('what is') || q.includes('tell me about')){
    const topic = q.replace(/who is|what is|tell me about|the|a |an /g,'').trim();
    const close = Object.keys(SNAKE_KB).find(k=>k.includes(topic)||topic.includes(k));
    if(close) return SNAKE_KB[close];
  }
  return 'Intel not found in local archive. Try: character names (Big Boss, Quiet, Ocelot), factions (Patriots, MSF, XOF), games (MGS1-4, MGSV), weapons (Metal Gear, FOXDIE), or themes (Les Enfants Terribles, S3 Plan, CQC). — Or switch to LIVE AI for deeper queries.';
}
